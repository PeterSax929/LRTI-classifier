06-differential_expr



library(edgeR)
library(limma)
library(dplyr)
library(tibble)
library(ggplot2)
library(ggrepel)






# 6.1. Import data


data_dir <- "C:/Users/User/Documents/UCSF/BREATH2.Gene/data"
raw_data_dir <- "C:/Users/User/Documents/UCSF/raw_data"
results_dir <- "C:/Users/User/Documents/UCSF/BREATH2.Gene/results"

metadata_df_with_folds <- read.csv(
  file.path(data_dir, "metadata_df_with_folds.csv"),
  stringsAsFactors=F
)

host_counts <- read.csv(
  file.path(raw_data_dir, "host_gene_counts.csv"),
  stringsAsFactors=F,
  row.names=1
)[, -1]

# Restrict to samples in metadata
counts <- host_counts[
  ,
  metadata_df_with_folds$sample_name,
  drop=F
]

stopifnot(
  all(
    colnames(counts) == metadata_df_with_folds$sample_name
  )
)






# 6.2. Filtering + TMM normalization


# Define group
group <- factor(
  metadata_df_with_folds$LRTI_adjudication,
  levels=c("No Evidence","Definite")
)

# Build DGEList
dge <- DGEList(
  counts=counts,
  group=group
)

# Apply filtering
keep <- rowSums(dge$counts > 10) > 0.2 * ncol(dge$counts)

dge <- dge[
  keep,
  ,
  keep.lib.sizes=F
]

# TMM normalization
dge <- calcNormFactors(
  dge,
  method="TMM"
)

# Check dimensions
dim(dge)
[1] 13941   105
# 6.3. voom transformation


# Ensure covariates are correct type
metadata_df_with_folds$sex <- factor(
  metadata_df_with_folds$sex
)

metadata_df_with_folds$age_in_years <- as.numeric(
  metadata_df_with_folds$age_in_years
)

# Design matrix with covariates
design <- model.matrix(
  ~ group + age_in_years + sex,
  data=metadata_df_with_folds
)

# Apply voom
v <- voom(
  dge,
  design,
  plot=T
)


# Sanity check
colnames(design)
[1] "(Intercept)"   "groupDefinite" "age_in_years"  "sexMale"      
dim(v$E)
[1] 13941   105
# 6.4. Linear modeling with limma


# Fit gene-wise linear models
fit <- lmFit(
  v,
  design
)

# Empirical Bayes moderation
fit <- eBayes(
  fit
)

# View coefficient names
colnames(coef(fit))
[1] "(Intercept)"   "groupDefinite" "age_in_years"  "sexMale"      
de_results <- topTable(
  fit,
  coef="groupDefinite",
  number=Inf,
  sort.by="P"
)

# Convert to data frame
de_results <- as.data.frame(de_results) %>%
  rownames_to_column("gene")






# 6.5. Format limma results for volcano plot


# Rename columns
res_df <- de_results %>%
  dplyr::rename(
    log2FoldChange=logFC,
    padj=adj.P.Val
  ) %>%
  mutate(
    padj=ifelse(
      is.na(padj),
      1,
      padj
    ),
    log2FoldChange=ifelse(
      is.na(log2FoldChange),
      0,
      log2FoldChange
    ),
    negLog10P=-log10(padj + 1e-300)
  )

# Map ENSG to gene symbol
res_df$gene_symbol <- ensg2gene$gene_symbol[
  match(
    res_df$gene,
    ensg2gene$X
  )
]

res_df$gene_symbol[
  is.na(res_df$gene_symbol)
] <- res_df$gene[
  is.na(res_df$gene_symbol)
]

# Significance flag
res_df <- res_df %>%
  mutate(
    sig=padj < 0.05
  )

# Biological direction
res_df <- res_df %>%
  mutate(
    direction=case_when(
      padj < 0.05 & log2FoldChange > 0 ~ "Definite",
      padj < 0.05 & log2FoldChange < 0 ~ "NoEvidence",
      TRUE ~ "NotSignificant"
    )
  )






# 6.6. Volcano Plot


# Select genes to label
label_data <- bind_rows(
  res_df %>%
    filter(direction == "Definite") %>%
    arrange(
      desc(abs(log2FoldChange)),
      padj
    ) %>%
    slice_head(n=10),

  res_df %>%
    filter(direction == "NoEvidence") %>%
    arrange(
      desc(abs(log2FoldChange)),
      padj
    ) %>%
    slice_head(n=10)
)


# Volcano plot
volcano_plot_gene <- ggplot(
  res_df,
  aes(
    x=log2FoldChange,
    y=negLog10P
  )
) +

  # Points
  geom_point(
    aes(color=direction),
    alpha=0.75,
    size=2
  ) +

  scale_color_manual(
    values=c(
      "Definite"="#D55E00",
      "NoEvidence"="#8B4789",
      "NotSignificant"="grey85"
    ),
    guide="none"
  ) +

  # FDR threshold
  geom_hline(
    yintercept=-log10(0.05),
    linetype="dashed",
    linewidth=0.6,
    color="grey40"
  ) +

  # Fold-change reference lines
  geom_vline(
    xintercept=c(-1,1),
    linetype="dotted",
    colour="grey70",
    linewidth=0.4
  ) +

  # Gene labels
  geom_text_repel(
    data=label_data,
    aes(label=gene_symbol),
    size=3.2,
    fontface="plain",
    max.overlaps=Inf,
    box.padding=0.7,
    point.padding=0.5,
    segment.color="grey60",
    segment.size=0.3
  ) +

  # Direction arrows
  annotate(
    "segment",
    x=-1,
    xend=-4,
    y=8,
    yend=8,
    linewidth=0.7,
    colour="#8B4789",
    arrow=arrow(
      length=unit(0.18,"cm"),
      type="closed"
    )
  ) +

  annotate(
    "text",
    x=-2.5,
    y=8.35,
    label="Downregulated in LRTI",
    size=3.8,
    colour="#8B4789"
  ) +

  annotate(
    "segment",
    x=1,
    xend=4,
    y=8,
    yend=8,
    linewidth=0.7,
    colour="#D55E00",
    arrow=arrow(
      length=unit(0.18,"cm"),
      type="closed"
    )
  ) +

  annotate(
    "text",
    x=2.5,
    y=8.35,
    label="Upregulated in LRTI",
    size=3.8,
    colour="#D55E00"
  ) +

  # DE gene count
  annotate(
    "text",
    x=0,
    y=7.1,
    label=paste0(
      format(
        sum(res_df$sig),
        big.mark=","
      ),
      " DE genes"
    ),
    size=3.8,
    family="Arial",
    colour="black"
  ) +

  # Theme
  theme_classic(
    base_size=12,
    base_family="Arial"
  ) +

  theme(
    text=element_text(
      family="Arial",
      size=12,
      face="plain"
    ),
    axis.title=element_text(
      family="Arial",
      size=12,
      colour="black",
      face="plain"
    ),
    axis.text=element_text(
      family="Arial",
      size=12,
      colour="black",
      face="plain"
    ),
    axis.line=element_line(
      colour="black",
      linewidth=0.5
    ),
    plot.title=element_text(
      family="Arial",
      size=12,
      face="plain",
      colour="black",
      hjust=0.5
    ),
    plot.margin=ggplot2::margin(
      10,
      15,
      10,
      10
    )
  ) +

  labs(
    x=expression(log[2]*" fold change"),
    y=expression(-log[10](P[adj])),
    title="Transcriptomic differential expression"
  ) +

  scale_y_continuous(
    expand=expansion(
      mult=c(0.02,0.15)
    )
  )


print(volcano_plot_gene)


# Save results
write.csv(
  res_df,
  file.path(
    results_dir,
    "DE_limma_results_transcriptome.csv"
  ),
  row.names=F
)


# Save editable vector PDF
