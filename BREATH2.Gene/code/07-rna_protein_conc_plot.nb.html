library(dplyr)
library(ggplot2)
library(ggrepel)
library(openxlsx)






# 7.1. Import data


results_dir <- "C:/Users/peter/OneDrive/Documents/UCSF/BREATH2.Gene/results"

rna_res <- read.csv(
  file.path(
    results_dir,
    "DE_limma_results_transcriptome.csv"
  ),
  stringsAsFactors = F
)

prot_res <- read.csv(
  file.path(
    "C:/Users/peter/OneDrive/Documents/UCSF/BREATH2.Prot/results",
    "DA_results_proteomics.csv"
  ),
  stringsAsFactors = F
)






# 7.2. Merge IDs


rna_df <- rna_res %>%
  dplyr::select(
    gene_symbol,
    log2FC_RNA = log2FoldChange,
    padj_RNA = padj
  ) %>%
  filter(
    !is.na(gene_symbol),
    gene_symbol != ""
  ) %>%
  distinct(
    gene_symbol,
    .keep_all = T
  )


prot_df <- prot_res %>%
  dplyr::select(
    gene_symbol = gene,
    log2FC_protein = log2FoldChange,
    padj_protein = padj
  ) %>%
  filter(
    !is.na(gene_symbol),
    gene_symbol != ""
  ) %>%
  distinct(
    gene_symbol,
    .keep_all = T
  )


merged_df <- inner_join(
  rna_df,
  prot_df,
  by = "gene_symbol"
)


merged_df <- merged_df %>%
  mutate(
    sig_class = case_when(
      padj_RNA < 0.05 & padj_protein < 0.05 ~ "RNA & Protein",
      padj_RNA < 0.05 ~ "RNA only",
      padj_protein < 0.05 ~ "Protein only",
      TRUE ~ "Not Significant"
    ),
    point_size = case_when(
      sig_class == "RNA & Protein" ~ 3.0,
      sig_class == "RNA only" ~ 1.8,
      sig_class == "Protein only" ~ 1.8,
      TRUE ~ 0.7
    )
  )


merged_df$sig_class <- factor(
  merged_df$sig_class,
  levels = c(
    "RNA only",
    "Protein only",
    "RNA & Protein",
    "Not Significant"
  )
)

# 7.4. Prepare plot content


label_df <- merged_df %>%
  filter(sig_class == "RNA & Protein") %>%
  mutate(
    concordance = abs(log2FC_RNA) + abs(log2FC_protein),
    quadrant = case_when(
      log2FC_RNA > 0 & log2FC_protein > 0 ~ "URQ",
      log2FC_RNA < 0 & log2FC_protein < 0 ~ "RLQ",
      TRUE ~ "Other"
    )
  ) %>%
  group_by(quadrant) %>%
  slice_max(
    concordance,
    n = 4
  ) %>%
  ungroup() %>%
  bind_rows(
    merged_df %>%
      filter(sig_class == "RNA & Protein") %>%
      mutate(
        concordance = sign(log2FC_RNA) == sign(log2FC_protein),
        combined_fc = abs(log2FC_RNA) + abs(log2FC_protein)
      ) %>%
      slice_max(
        combined_fc,
        n = 12
      )
  ) %>%
  distinct(
    gene_symbol,
    .keep_all = TRUE
  )






# 7.5. RNA vs. Protein concordance plot


concordance_plot <- ggplot(
  merged_df,
  aes(
    x = log2FC_RNA,
    y = log2FC_protein,
    color = sig_class
  )
) +

  geom_point(
    data = merged_df %>%
      filter(sig_class == "Not Significant"),
    aes(size = point_size),
    alpha = 0.18
  ) +

  geom_point(
    data = merged_df %>%
      filter(sig_class != "Not Significant"),
    aes(size = point_size),
    alpha = 0.90
  ) +

  scale_size_identity() +

  geom_text_repel(
    data = label_df,
    aes(label = gene_symbol),
    size = 3.2,
    fontface = "plain",
    box.padding = 0.5,
    segment.color = "grey60",
    segment.size = 0.3,
    max.overlaps = 25,
    show.legend = FALSE
  ) +

  geom_hline(
    yintercept = 0,
    linetype = "dotted",
    color = "grey70",
    linewidth = 0.4
  ) +

  geom_vline(
    xintercept = 0,
    linetype = "dotted",
    color = "grey70",
    linewidth = 0.4
  ) +

  scale_color_manual(
    values = c(
      "RNA & Protein" = "purple",
      "RNA only" = "#0072B2",
      "Protein only" = "#009E73",
      "Not Significant" = "grey85"
    ),
    breaks = c(
      "RNA & Protein",
      "RNA only",
      "Protein only"
    )
  ) +
  
  scale_y_continuous(
    limits = c(-5, 5), 
    labels = function(x) sprintf("%.1f", x)
  ) +

  theme_classic(
    base_size = 16,
    base_family = "Arial"
  ) +

  labs(
    x = expression("Transcriptome log"[2] * " fold change"),
    y = expression("Proteome log"[2] * " fold change"),
    title = "Concordance of LRTI-associated features",
    color = NULL
  ) +

  theme(
    text = element_text(
      family = "Arial",
      size = 12,
      face = "plain"
    ),
    axis.title = element_text(
      family = "Arial",
      size = 16,
      colour = "black",
      face = "plain"
    ),
    axis.text = element_text(
      family = "Arial",
      size = 16,
      colour = "black",
      face = "plain"
    ),
    axis.line = element_line(
      colour = "black",
      linewidth = 0.5
    ),
    plot.title = element_text(
      family = "Arial",
      size = 16,
      face = "plain",
      colour = "black",
      hjust = 0.5
    ),
    plot.margin = ggplot2::margin(
      10,
      15,
      10,
      10,
    ),
    legend.position = c(0.5, 0.08),
    legend.direction = "horizontal",
    legend.title = element_blank(),
    legend.text = element_text(
      family = "Arial",
      size = 16,
      face = "plain"
    ),
    legend.key.size = unit(
      1.2,
      "lines"
    )
  )


print(concordance_plot)


ggsave(
