library(dplyr)

# 13.1 Import Data


data_dir    <- "C:/Users/User/Documents/UCSF/BREATH2.Gene/data"

# Gene matrix
full_vsd <- read.csv(
  file.path(data_dir, "vst_full.csv"),
  stringsAsFactors=F, check.names=F, row.names=1)

full_vsd <- full_vsd %>%
  rownames_to_column("gene") %>%
  mutate(
    gene_symbol = ensg2gene[gene],
    gene_symbol = ifelse(is.na(gene_symbol), gene, gene_symbol))

  file.path(

  stringsAsFactors=F)

  file.path(data_dir, "metadata_df_with_folds.csv"),

# 13.2 Find shared analytes


# Shared analytes
shared_genes <- intersect(full_vsd$gene_symbol, colnames(protein_mat)[-1])

cat("Shared analytes:", length(shared_genes), "\n")
Shared analytes: 6247 
genes_biomarker <- c("FABP4", "ISG15", "IFIT2", "IFIT3", "APOE", "SLAMF7", "OLAH")





# 13.3 Compute per-analyte Pearson r


cor_per_gene <- full_vsd %>%
  filter(gene_symbol %in% shared_genes) %>%
  pivot_longer(cols=-c(gene, gene_symbol), names_to="sample_name", values_to="expr_rna") %>%
  inner_join(metadata_df %>% dplyr::select(sample_name, patient_id), by="sample_name") %>%
  inner_join(
    protein_mat %>% { colnames(.)[1] <- "patient_id"; . } %>%
      mutate(patient_id = as.integer(patient_id)) %>%
      pivot_longer(cols=-patient_id, names_to="gene_symbol", values_to="expr_protein") %>%
      filter(gene_symbol %in% shared_genes),
    by = c("patient_id", "gene_symbol")
  ) %>%
  group_by(gene_symbol) %>%
  summarize(
    pearson_r = cor.test(expr_rna, expr_protein, method="pearson")$estimate,
    p_value   = cor.test(expr_rna, expr_protein, method="pearson")$p.value,
    .groups="drop"
  ) %>%
  mutate(padj = p.adjust(p_value, method="BH"))

  arrange(desc(pearson_r))






# 13.4 Density plot of per-analyte Pearson r


genes_lasso <- unique(c(lasso_coefs_gene$gene_symbol, lasso_coefs_prot$protein))

cor_per_gene <- cor_per_gene %>%
  mutate(
    category = case_when(
      gene_symbol %in% genes_biomarker ~ "Biomarker",
      gene_symbol %in% genes_lasso     ~ "LASSO-selected",
      TRUE                             ~ "Background"
    )
  )

genes_highlight <- cor_per_gene %>% filter(category != "Background")
genes_label     <- cor_per_gene %>% filter(category == "Biomarker")

ggplot(cor_per_gene, aes(x = pearson_r)) +
  geom_density(
    fill = "#4C72B0", alpha = 0.25, color = "grey25", linewidth = 1) +
  geom_vline(
    xintercept = 0, linetype = "solid", color = "black", linewidth = 0.8) +
  geom_vline(
    xintercept = median(cor_per_gene$pearson_r, na.rm = TRUE),
    linetype = "dashed", color = "black", linewidth = 0.7
  ) +
  # LASSO-selected rug ticks
  geom_rug(
    data = genes_highlight %>% filter(category == "LASSO-selected"),
    aes(x = pearson_r),
    color = "#E69F00", linewidth = 0.5, alpha = 0.7, length = unit(0.04, "npc")
  ) +
  # Biomarker rug ticks
  geom_rug(
    data = genes_highlight %>% filter(category == "Biomarker"),
    aes(x = pearson_r),
    color = "red", linewidth = 0.7, alpha = 0.8, length = unit(0.05, "npc")
  ) +
  # Biomarker labels rotated above rug
  geom_text(
    data = genes_label,
    aes(x = pearson_r, y = 0.12, label = gene_symbol),
    angle = 90, hjust = 0, size = 3, color = "red", fontface = "bold",
    check_overlap = TRUE
  ) +
  annotate("text",
    x = median(cor_per_gene$pearson_r, na.rm=T) + 0.03,
    y = Inf,
    label = paste0("Median r = ", round(median(cor_per_gene$pearson_r), 3)),
    hjust = 0, vjust = 1.5, size = 3.5, fontface = "bold"
  ) +
  scale_x_continuous(limits = c(-0.5, 0.9), breaks = seq(-0.4, 0.8, 0.2)) +
  scale_color_identity() +
  labs(
    title = "RNA-Protein Concordance Across Shared Analytes",
    subtitle = paste0(
      "n = ", nrow(cor_per_gene), " shared analytes | ",
      "orange = LASSO-selected | red = 7 biomarker genes"
    ),
    x = "Pearson r (RNA vs Protein)",
    y = "Density"
  ) +
  theme_classic(base_size = 13) +
  theme(
    text = element_text(family = "Helvetica"),
    axis.text = element_text(face = "bold"),
    axis.title = element_text(face = "bold"),
    plot.title = element_text(face = "bold", hjust = 0.5),
    plot.subtitle = element_text(hjust = 0.5),
    panel.grid = element_blank()
  )

# 13.5 Density plot split by LRTI adjudication


cor_per_gene_adj <- full_vsd %>%
  filter(gene_symbol %in% shared_genes) %>%
  pivot_longer(cols=-c(gene, gene_symbol), names_to="sample_name", values_to="expr_rna") %>%
  inner_join(metadata_df %>% dplyr::select(sample_name, patient_id, LRTI_adjudication), by="sample_name") %>%
  inner_join(
    protein_mat %>% { colnames(.)[1] <- "patient_id"; . } %>%
      mutate(patient_id = as.integer(patient_id)) %>%
      pivot_longer(cols=-patient_id, names_to="gene_symbol", values_to="expr_protein") %>%
      filter(gene_symbol %in% shared_genes),
    by = c("patient_id", "gene_symbol")
  ) %>%
  group_by(gene_symbol, LRTI_adjudication) %>%
  summarize(pearson_r = cor(expr_rna, expr_protein, method="pearson", use="complete.obs"),
            .groups="drop") %>%
  mutate(
    category = case_when(
      gene_symbol %in% genes_biomarker ~ "Biomarker",
      gene_symbol %in% genes_lasso     ~ "LASSO-selected",
      TRUE                             ~ "Background"
    ),
    LRTI_adjudication = factor(LRTI_adjudication, levels = c("Definite", "No Evidence"))
  )

# Separate label dfs per group
labels_definite    <- cor_per_gene_adj %>% filter(category == "Biomarker", LRTI_adjudication == "Definite")
labels_noevidence  <- cor_per_gene_adj %>% filter(category == "Biomarker", LRTI_adjudication == "No Evidence")
lasso_definite     <- cor_per_gene_adj %>% filter(category == "LASSO-selected", LRTI_adjudication == "Definite")
lasso_noevidence   <- cor_per_gene_adj %>% filter(category == "LASSO-selected", LRTI_adjudication == "No Evidence")

# Plot function to avoid duplication
plot_density_adj <- function(data_df, lasso_df, label_df, adj_label, fill_color) {
  ggplot(data_df, aes(x = pearson_r)) +
    geom_density(
      fill = fill_color, alpha = 0.25, color = "grey25", linewidth = 1) +
    geom_vline(
      xintercept = 0, linetype = "solid", color = "black", linewidth = 0.8) +
    geom_vline(
      xintercept = median(data_df$pearson_r, na.rm = TRUE),
      linetype = "dashed", color = "black", linewidth = 0.7
    ) +
    geom_rug(
      data = lasso_df,
      aes(x = pearson_r),
      color = "#E69F00", linewidth = 0.5, alpha = 0.7, length = unit(0.05, "npc"),
      inherit.aes = FALSE
    ) +
    geom_rug(
      data = label_df,
      aes(x = pearson_r),
      color = "red", linewidth = 0.7, alpha = 0.8, length = unit(0.07, "npc"),
      inherit.aes = FALSE
    ) +
    geom_text(
      data = label_df,
      aes(x = pearson_r, y = 0.12, label = gene_symbol),
      angle = 90, hjust = 0, size = 3, color = "red", fontface = "bold",
      check_overlap = TRUE
    ) +
    annotate("text",
      x = median(data_df$pearson_r, na.rm=T) + 0.03,
      y = Inf,
      label = paste0("Median r = ", round(median(data_df$pearson_r, na.rm=T), 3)),
      hjust = 0, vjust = 1.5, size = 3.5, fontface = "bold"
    ) +
    scale_x_continuous(limits = c(-0.5, 0.9), breaks = seq(-0.4, 0.8, 0.2)) +
    labs(
      title = paste0("RNA-Protein Concordance — ", adj_label),
      subtitle = paste0("n = ", nrow(data_df), " shared analytes | orange = LASSO-selected | red = 7 biomarker genes"),
      x = "Pearson r (RNA vs Protein)",
      y = "Density"
    ) +
    theme_classic(base_size = 13) +
  theme(
    text = element_text(family = "Helvetica"),
    axis.text = element_text(face = "bold"),
    axis.title = element_text(face = "bold"),
    plot.title = element_text(face = "bold", hjust = 0.5),
    plot.subtitle = element_text(hjust = 0.5),
    panel.grid = element_blank()
  )
}

p_definite <- plot_density_adj(
  data_df   = cor_per_gene_adj %>% filter(LRTI_adjudication == "Definite"),
  lasso_df  = lasso_definite,
  label_df  = labels_definite,
  adj_label = "Definite LRTI",
  fill_color = "red"
)

p_noevidence <- plot_density_adj(
  data_df   = cor_per_gene_adj %>% filter(LRTI_adjudication == "No Evidence"),
  lasso_df  = lasso_noevidence,
  label_df  = labels_noevidence,
  adj_label = "No Evidence",
  fill_color = "#0072B2"
)
