# 5.2. LASSO Workflow


# Function for a single outer CV fold
lasso_cv_outer_fold <- function(test_fold) {
  set.seed(8634782 + test_fold)
  
  train <- patient_info_with_folds$fold != test_fold
  
  # Fit full lambda path
  fit <- glmnet::glmnet(
    X[train, ], y[train],
    family = "binomial",
    standardize = TRUE
    )

  




# 5.2a. Select lambda based on LASSO CV


  # Perform cross validated LASSO to pick optimal lambda
  cv_fit <- cv.glmnet(
    X[train, ], y[train],
    family = "binomial",
    standardize = TRUE,
    nfolds = nfolds
    )

  # Predictions on test fold
  preds <- predict(fit, X[!train, ], type = "response",
    s = cv_fit$lambda.1se)[, 1] %>%
    data.frame(pred = .) %>%
    tibble::rownames_to_column("sample_name")

  # Extract coefficients at selected lambda
  coefs_vec <- coef(fit, s = cv_fit$lambda.1se)[-1, 1]  # remove intercept

  # Keep nonzero coefficients only
  nonzero <- coefs_vec[coefs_vec != 0]






# 5.2b. Build data frame


  # Build data frame
  nonzero <- data.frame(
    protein = names(nonzero),
    coef = as.numeric(nonzero),
    fold = test_fold
    )

  list(pred = preds, mod = fit, coef = nonzero) 
}

# Alt 5.2. LASSO with fixed protein count per fold


# Set target number of proteins (e.g. 1 or 3)
target_n <- 1

lasso_cv_fixed_n <- function(test_fold) {
  set.seed(8634782 + test_fold)
  
  
  
  
  
  
# Alt 5.2a. Select lambda
  

  train <- patient_info_with_folds$fold != test_fold

  # Fit full lambda path
  fit <- glmnet::glmnet(
    X[train, ], y[train],
    family = "binomial",
    standardize = TRUE
  )

  # Cross validated LASSO to pick optimal lambda
  cv_fit <- cv.glmnet(
    X[train, ], y[train],
    family = "binomial",
    standardize = TRUE,
    nfolds = nfolds
  )

  # Count nonzero coefficients at each lambda
  nz <- apply(fit$beta, 2, function(b) sum(b != 0))

  # Find lambda that selects exactly target_n proteins
  if (any(nz == target_n)) {
    lambda_target <- fit$lambda[which(nz == target_n)[1]]
  } else {
    # Use closest lambda to target_n
    lambda_target <- fit$lambda[which.min(abs(nz - target_n))]
  }

  # Extract coefficients at selected lambda
  coefs_vec <- coef(fit, s = lambda_target)[-1, 1]
  nonzero <- coefs_vec[coefs_vec != 0]

  # Trim to top target_n by absolute value
  if (length(nonzero) > target_n) {
    nonzero <- nonzero[order(abs(nonzero), decreasing=TRUE)[1:target_n]]
  }

  # Adjust lambda until target_n reached (safeguard)
  if (length(nonzero) < target_n) {
    lam_idx <- which.min(abs(fit$lambda - lambda_target))
    while (length(nonzero) < target_n && lam_idx < length(fit$lambda)) {
      lam_idx <- lam_idx + 1
      coefs_vec <- coef(fit, s = fit$lambda[lam_idx])[-1, 1]
      nonzero <- coefs_vec[coefs_vec != 0]
    }
    if (length(nonzero) > target_n) {
      nonzero <- nonzero[order(abs(nonzero), decreasing=TRUE)[1:target_n]]
    }
  }

  # Safeguard
  stopifnot(length(nonzero) == target_n)

  # Predictions
  preds <- predict(fit, X[!train, ], type="response",
    s = lambda_target)[, 1] %>%
    data.frame(pred = .) %>%
    tibble::rownames_to_column("sample_name")

  
  
  
  
  
# Alt 5.2b. Build data frame
  
  # Coefs data frame
  nonzero_df <- data.frame(
    protein = names(nonzero),
    coef    = as.numeric(nonzero),
    fold    = test_fold
  )

  list(pred = preds, mod = fit, coef = nonzero_df)
}
